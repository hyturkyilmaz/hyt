# --- O-02: istek içi doğrulamada Retry-After 60 sn'ye kadar uyulur, kova beklemesi sınırsız -----


def test_poc_validation_request_sleeps_retry_after_inside_http_request(
    make_ty_api: Callable[..., Api], trendyol: FakeTrendyol
) -> None:
    api = make_ty_api()
    api.signup_and_login("o@example.com", tenant="Mağaza")
    trendyol.status = 429

    assert r.status_code == 503

    assert trendyol.fake.sleeps == [60.0]


async def test_poc_shared_limiter_makes_other_tenant_wait_after_429() -> None:
    """Paylaşılan kova satıcı kimliğine bağlı: bir 429 başka kiracının doğrulamasını da bekletir."""
    fake = FakeTime()
    shared = RateLimiterRegistry(clock=fake.monotonic, sleep=fake.sleep)
    calls = {"n": 0}

    def handler(request: httpx.Request) -> httpx.Response:
        calls["n"] += 1
        if calls["n"] == 1:
            return json_response({}, 429, {"Retry-After": "60"})
        return json_response(load_fixture("inventory_and_price.json"))

    first, _ = make_client(handler, fake=fake, limiter=shared, config=TrendyolConfig(max_retries=0))
    async with first:
        with pytest.raises(TrendyolRateLimitedError):
            await first.verify_connection()
    before = list(fake.sleeps)
    # Aynı satıcı kimliği, başka kiracı/anahtar (aynı süreç içi kayıt): gönderimden önce 60 sn bekler.
    second, _ = make_client(handler, fake=fake, limiter=shared, config=TrendyolConfig(max_retries=0))
    async with second:
        await second.verify_connection()
    assert fake.sleeps[len(before) :] == [60.0]


# --- D-01: PII, hata nesnesinin traceback çerçeve yerellerinde kalıyor ------------------------


async def test_poc_raw_pii_survives_in_traceback_frame_locals() -> None:
    raw = load_fixture("shipment_packages_v2.json")
    raw["content"][0]["shipmentPackageId"] = "sayı-değil"  # şema hatası tetiklenir
    client, _ = make_client(lambda request: json_response(raw))
    async with client:
        with pytest.raises(TrendyolProtocolError) as info:
            await client.get_shipment_packages()
    # Sentry include_local_variables=True (varsayılan) benzeri: çerçeve yerellerinin repr'i
    tb = info.value.__traceback__
    dumped = []
    while tb is not None:
        # Yalnız istemci çerçeveleri (testin kendi `raw` yereli sayılmaz)
        if tb.tb_frame.f_code.co_filename.endswith("integrations/trendyol/client.py"):
            dumped.append(repr(tb.tb_frame.f_locals))
        tb = tb.tb_next
    assert len(dumped) == 3  # get_shipment_packages, _page, _validate
    text = " ".join(dumped)
    leaked = [s for s in PII_SENTINELS if s in text]
    assert leaked, "çerçeve yerellerinde ham PII yok"
    # Hata nesnesinin kendisi temiz (mevcut testlerin doğruladığı kısım)
    assert not any(s in f"{info.value!r} {info.value.args}" for s in PII_SENTINELS)


# --- D-03: httpx.DecodingError ve RecursionError TrendyolError'a eşlenmiyor --------------------


async def test_poc_decoding_error_escapes_error_mapping() -> None:
    def handler(request: httpx.Request) -> httpx.Response:
        return httpx.Response(
            200, stream=httpx.ByteStream(b"gzip-degil"), headers={"Content-Encoding": "gzip"}
        )

    client, _ = make_client(handler)
    async with client:
        with pytest.raises(httpx.DecodingError) as info:
            await client.verify_connection()
    assert not isinstance(info.value, TrendyolError)
    # İstisna istek nesnesini (Authorization başlığı dahil) taşır; httpx repr'i maskeler.
    assert "Authorization" in info.value.request.headers


async def test_poc_deeply_nested_json_raises_recursion_error_not_protocol_error() -> None:
    deep = b'{"a":' + b"[" * 100_000 + b"]" * 100_000 + b"}"

    client, _ = make_client(lambda request: httpx.Response(200, content=deep))
    async with client:
        with pytest.raises(RecursionError):
            await client.verify_connection()

# --- D-02: AAD kiracıyı ve satıcıyı bağlamıyor -----------------------------------------------


def test_poc_aad_does_not_bind_tenant_or_seller() -> None:
    cipher = FieldCipher(base64.b64decode(TEST_KEY))
    account_id = uuid4()
    original = MarketplaceAccount(id=account_id, tenant_id=uuid4(), seller_id="1111")
    original.api_key_encrypted = encrypt_secret(cipher, account_id, "api_key", SecretStr("AnahtarABCDEFGH"))
    original.api_secret_encrypted = encrypt_secret(cipher, account_id, "api_secret", SecretStr("SirABCDEFGH"))
    # Aynı satır kimliği, farklı kiracı ve satıcı: çözme başarılı (AAD yalnız id + alan adı)
    moved = MarketplaceAccount(
        id=account_id,
        tenant_id=uuid4(),
        seller_id="2222",
        api_key_encrypted=original.api_key_encrypted,
        api_secret_encrypted=original.api_secret_encrypted,
    )
    creds = decrypt_credentials(cipher, moved)
    assert creds.seller_id == "2222" and creds.api_key.get_secret_value() == "AnahtarABCDEFGH"


# --- D-04: netkar_app şifreli sütunları okuyabilir, kimlik sütunlarını güncelleyebilir -----------


def test_poc_app_role_column_privileges(admin_conn: psycopg.Connection) -> None:
    rows = admin_conn.execute(
        "SELECT column_name, has_column_privilege('netkar_app', 'marketplace_accounts', column_name, 'SELECT'),"
        " has_column_privilege('netkar_app', 'marketplace_accounts', column_name, 'UPDATE') "
        "FROM information_schema.columns WHERE table_name = 'marketplace_accounts'"
    ).fetchall()
    priv = {name: (sel, upd) for name, sel, upd in rows}
    assert priv["api_key_encrypted"] == (True, True)
    assert priv["api_secret_encrypted"] == (True, True)
    for col in ("tenant_id", "seller_id", "platform", "created_by", "created_at", "api_key_last4"):
        assert priv[col][1] is True, col

    make_ty_api: Callable[..., Api], admin_conn: psycopg.Connection

    api = make_ty_api()

    ).fetchall()
    assert any(before[0][3:20] in r[0] for r in raw), "ölü sürümde şifreli değer yok"
    admin_conn.execute("DROP EXTENSION pageinspect")

# --- Kod kalitesi: totalPages yoksa yineleyiciler ilk sayfada sessizce durur ----------------------


async def test_poc_iterator_stops_after_first_page_when_total_pages_missing() -> None:
    pages = load_fixture("shipment_packages_v2.json")
    calls = {"n": 0}

    def handler(request: httpx.Request) -> httpx.Response:
        calls["n"] += 1
        data = json.loads(json.dumps(pages))
        data.pop("totalPages", None)
        data["totalElements"] = 5000
        return json_response(data)

    from datetime import UTC, datetime, timedelta

    t0 = datetime(2026, 10, 1, tzinfo=UTC)
    client, _ = make_client(handler)
    async with client:
        got = [p async for p in client.iter_shipment_packages(t0, t0 + timedelta(days=1))]
    assert calls["n"] == 1 and len(got) == len(pages["content"])

    api = make_ty_api()

    assert trendyol.requests == []
